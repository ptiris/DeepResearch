138class BudgetState:
139    """Tracks search budget consumption and pressure level.
140    
141    Attributes
142    ----------
143    effective_search_calls : int
144        Number of search calls executed.
145    search_call_budget : int
146        Maximum allowed search calls.
147    observation_tokens_used : int
148        Tokens consumed by observations.
149    observation_token_budget : int
150        Maximum observation tokens allowed.
151    current_turn : int
152        Current conversation turn.
153    max_turns : int
154        Maximum conversation turns allowed.
155    """
156    effective_search_calls: int = 0
157    search_call_budget: int = 10
158    observation_tokens_used: int = 0
159    observation_token_budget: int = 8000
160    current_turn: int = 0
161    max_turns: int = 10
162
163    @property
164    def pressure(self) -> float:
165        """Compute overall budget pressure as max of normalized metrics.
166
167        Returns
168        -------
169        float
170            Pressure ratio in [0, 1+], where higher means more constrained.
171        """
172        return max(
173            self.effective_search_calls / max(self.search_call_budget, 1),
174            self.observation_tokens_used / max(self.observation_token_budget, 1),
175            self.current_turn / max(self.max_turns, 1),
176        )
177
178    @property
179    def pressure_level(self) -> str:
180        """Categorize pressure level for policy decisions.
181
182        Returns
183        -------
184        str
185            'low' (<0.4), 'medium' (0.4-0.7), or 'high' (>0.7).
186        """
187        p = self.pressure
188        if p < 0.4:
189            return "low"
190        elif p < 0.7:
191            return "medium"
192        return "high"

283    def find_similar(
284        self,
285        query_embedding: list[float],
286        top_k: int = 5,
287    ) -> list[tuple[SearchMemoryEntry, float]]:
288        """Find top-k similar historical queries by cosine similarity.
289
290        Parameters
291        ----------
292        query_embedding : list[float]
293            Query vector to compare against.
294        top_k : int, optional
295            Maximum number of results to return.
296
297        Returns
298        -------
299        list[tuple[SearchMemoryEntry, float]]
300            List of (entry, similarity_score) tuples, sorted by score descending.
301        """
302        scored = [
303            (entry, _cosine_similarity(query_embedding, entry.query_embedding))
304            for entry in self.entries
305        ]
306        scored.sort(key=lambda x: x[1], reverse=True)
307        return scored[:top_k]

412def _cosine_similarity(a: list[float], b: list[float]) -> float:
413    """Compute cosine similarity between two vectors.
414
415    Parameters
416    ----------
417    a : list[float]
418        First vector.
419    b : list[float]
420        Second vector.
421
422    Returns
423    -------
424    float
425        Cosine similarity score in [-1, 1]. Returns 0.0 if either vector is zero.
426    """
427    dot = sum(x * y for x, y in zip(a, b))
428    norm_a = math.sqrt(sum(x * x for x in a))
429    norm_b = math.sqrt(sum(x * x for x in b))
430    if norm_a == 0 or norm_b == 0:
431        return 0.0
432    return dot / (norm_a * norm_b)

435class SearchController:
436    """Coordinates search execution with caching, deduplication, and budget management.
437    
438    The SearchController decides how to handle a list of search queries:
439    - Which queries to execute vs. reuse from cache
440    - Which queries to merge together
441    - How many results to retrieve based on budget pressure
442    
443    Parameters
444    ----------
445    params : dict or None
446        Configuration parameters for thresholds and cost estimates.
447        - high_similarity: Threshold for cache reuse (default 0.90)
448        - medium_similarity: Threshold for merge detection (default 0.75)
449        - pointer_turn_distance: Max turns to look back for cache (default 3)
450        - cost_per_search: Estimated cost per search call (default 0.01)
451        - cost_per_obs_token: Cost per observation token (default 0.00001)
452        - default_topk: Default number of results (default 10)
453    embed_fn : callable or None
454        Callable[[str], list[float]] for query embedding. If None,
455        similarity computation is skipped.
456    context_profile : ContextProfile or None
457        ContextProfile for admission and replay policies.
458    """
459    
460    def __init__(
461        self,
462        params: dict | None = None,
463        embed_fn=None,
464        context_profile: ContextProfile | None = None,
465    ):
466        
467        self.params = params or {}
468        self.embed_fn = embed_fn  # Callable[[str], list[float]]
469        self.context_profile = context_profile or ContextProfile(
470            policy="append_history",
471            replay_factor=2.0,
472            admission_policy="pass_through",
473            cache_replay_policy="pointer",
474        )
475
476        self.high_sim = self.params.get("high_similarity", 0.90)
477        self.medium_sim = self.params.get("medium_similarity", 0.75)
478        self.pointer_turn_dist = self.params.get("pointer_turn_distance", 3)
479
480        self.cost_per_search = self.params.get("cost_per_search", 0.01)
481        self.cost_per_obs_token = self.params.get("cost_per_obs_token", 0.00001)
482        self.estimated_obs_tokens = self.params.get("estimated_obs_tokens", 800)
483        self.estimated_replay_factor = self.params.get("estimated_replay_factor", 2.0)
484        self.latency_cost_coeff = self.params.get("latency_cost_coeff", 0.001)
485        self.estimated_latency_ms = self.params.get("estimated_latency_ms", 500.0)
486        self.cost_weight = self.params.get("cost_weight", 1.0)
487
488        # Section 10.1: pre-search utility weights
489        self.w_new_query = self.params.get("w_new_query", 1.0)
490        self.w_no_cache = self.params.get("w_no_cache", 0.5)
491        self.w_redundancy = self.params.get("w_redundancy", 0.8)
492        self.w_repeated_failure = self.params.get("w_repeated_failure", 0.6)

498    def pre_search(
499        self,
500        request: SearchRequest,
501        state: ControllerState,
502    ) -> PreSearchDecision:
503        """Plan search execution strategy for a list of queries.
504
505        Analyzes query similarity, checks cache for matches, and decides
506        which queries to execute, merge, or serve from cache based on
507        budget pressure and similarity thresholds.
508
509        Parameters
510        ----------
511        request : SearchRequest
512            The SearchRequest containing the queries to plan.
513        state : ControllerState
514            Current ControllerState with memory and budget.
515
516        Returns
517        -------
518        PreSearchDecision
519            PreSearchDecision containing query blocks with assigned actions.
520        """
521        query_list = request.query_list
522        n = len(query_list)
523
524        embeddings = self._embed_queries(query_list)
525        intra_sim = self._compute_intra_similarity(embeddings)
526        print(f"[Search Controller] Query similarity matrix for {n} queries:")
527        for i in range(n):
528            print(f"  Query[{i}]: {query_list[i][:50]}...")
529            for j in range(n):
530                print(f"    sim[{i}][{j}] = {intra_sim[i][j]:.4f}")
531        self.last_intra_sim = intra_sim
532        cache_hits = [
533            state.memory.find_similar(emb, top_k=3) for emb in embeddings
534        ]
535        pressure = state.budget_state.pressure
536        pressure_level = state.budget_state.pressure_level
537
538        assigned: set[int] = set()
539        blocks: list[QueryBlock] = []
540
541
542        for i in range(n):
543            if i in assigned:
544                continue
545
546            merge_group = [i]
547            for k in range(i + 1, n):
548                if k not in assigned and intra_sim[i][k] >= self.high_sim:
549                    merge_group.append(k)
550
551            if len(merge_group) > 1:
552                blocks.append(QueryBlock(
553                    queries=[query_list[idx] for idx in merge_group],
554                    original_indices=merge_group,
555                    action=SearchAction.MERGE,
556                    reason=f"high intra-call similarity among queries {merge_group}",
557                ))
558                assigned.update(merge_group)
559                continue
560
561            decision = self._decide_single_query(
562                index=i,
563                query_list=query_list,
564                intra_sim=intra_sim,
565                cache_hits=cache_hits,
566                pressure=pressure,
567                pressure_level=pressure_level,
568                budget_state=state.budget_state,
569            )
570
571            blocks.append(QueryBlock(
572                queries=[query_list[i]],
573                original_indices=[i],
574                action=decision.action,
575                reason=decision.reason,
576                cache_entry=decision.cache_entry,
577                adjusted_topk=decision.adjusted_topk,
578            ))
579            assigned.add(i)
580        print("[Search Controller] Final query blocks are")
581        for block in blocks :
582            print(f"Block {i} with query : {block.queries}")
583            print(f"Actions: {block.action}")
584        return PreSearchDecision(
585            task_id=request.task_id,
586            turn_id=request.turn_id,
587            query_blocks=blocks,
588        )

609    def _compute_intra_similarity(
610        self, embeddings: list[list[float]]
611    ) -> list[list[float]]:
612        """Compute pairwise cosine similarity matrix between query embeddings.
613        
614        Parameters
615    ----------
616    embeddings : list[list[float]]
617        List of query embedding vectors.
618
619    Returns
620    -------
621    list[list[float]]
622        NxN similarity matrix where sim[i][j] = similarity between i and j.
623        """
624        n = len(embeddings)
625        sim = [[0.0] * n for _ in range(n)]
626        for i in range(n):
627            for j in range(i + 1, n):
628                s = _cosine_similarity(embeddings[i], embeddings[j])
629                sim[i][j] = s
630                sim[j][i] = s
631        return sim

655    def _estimate_pre_cost(self) -> float:
656        """Estimate pre-search cost.
657
658        Returns
659        -------
660        float
661            Estimated cost in arbitrary units for executing a single query.
662
663        Formula: C_pre = p_search + n_obs*p_in + r_replay*n_obs*p_in + mu*t_search
664        """

666        return (
667            self.cost_per_search
668            + n_obs * self.cost_per_obs_token
669            + self.estimated_replay_factor * n_obs * self.cost_per_obs_token
670            + self.latency_cost_coeff * self.estimated_latency_ms
671        )

691        intra_sim : list[list[float]]
692            Pre-computed similarity matrix between queries.
693        cache_hits : list[list[tuple[SearchMemoryEntry, float]]]
694            Cache hit candidates from memory lookup.
695        pressure : float
696            Current budget pressure ratio.
697        pressure_level : str
698            'low', 'medium', or 'high'.
699        budget_state : BudgetState
700            Current BudgetState for decisions.
701
702        Returns
703        -------
704        QueryDecision
705            QueryDecision with the chosen action and reasoning.
706        """
707        query = query_list[index]
708
709        best_entry, best_sim = self._best_cache_hit(cache_hits[index])
710
711        if best_entry is not None and best_sim >= self.high_sim:
712            turn_dist = budget_state.current_turn - best_entry.turn_id # Forget when turn_dist is sometimes too far
713
714            if best_entry.tool_status == ToolStatus.SUCCESS_NONEMPTY:
715                return QueryDecision(
716                    query=query,
717                    original_index=index,
718                    action=SearchAction.REUSE_CACHE,
719                    reason=f"cache hit (sim={best_sim:.3f}, turn_dist={turn_dist}) with successful query: '{best_entry.query}'",
720                    cache_entry=best_entry,
721                )
722
723            if best_entry.tool_status in (ToolStatus.EMPTY, ToolStatus.FAILED):
724                if best_sim >= 0.95:
725                    return QueryDecision(
726                        query=query,
727                        original_index=index,
728                        action=SearchAction.EXECUTE,
729                        reason=f"near-exact repeat of {best_entry.tool_status.value} query (sim={best_sim:.3f}): '{best_entry.query}'",
730                        cache_entry=best_entry,
731                    )
732                
733
734        if pressure_level == "high" and self.context_profile.admission_policy == "budgeted":
735            for j in range(index):
736                if self.medium_sim <= intra_sim[index][j] < self.high_sim:
737                    adjusted_topk = self._reduce_topk(budget_state)
738                    return QueryDecision(
739                        query=query,
740                        original_index=index,
741                        action=SearchAction.REDUCE_TOPK,
742                        reason=f"medium intra-call similarity ({intra_sim[index][j]:.3f}) with query[{j}] under high budget pressure",
743                        adjusted_topk=adjusted_topk,
744                    )
745
746            adjusted_topk = self._reduce_topk(budget_state)
747            return QueryDecision(
748                query=query,
749                original_index=index,
750                action=SearchAction.REDUCE_TOPK,
751                reason=f"no cache hit, high budget pressure ({pressure:.2f})",
752                adjusted_topk=adjusted_topk,
753            )
754
755        return QueryDecision(
756            query=query,
757            original_index=index,
758            action=SearchAction.EXECUTE,
759            reason="no cache hit, normal execution",
760        )

728                        original_index=index,
729                        action=SearchAction.EXECUTE,
730                        reason=f"near-exact repeat of {best_entry.tool_status.value} query (sim={best_sim:.3f}): '{best_entry.query}'",
731                        cache_entry=best_entry,
732                    )
733                
734
735        if pressure_level == "high" and self.context_profile.admission_policy == "budgeted":
